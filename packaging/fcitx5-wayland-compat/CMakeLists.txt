# Preserve the original source list and generated header naming, with the
# installed scanner instead of requiring all of Fcitx's unrelated build stack.
function(ecm_add_wayland_client_protocol output)
  cmake_parse_arguments(P "" "PROTOCOL;BASENAME" "" ${ARGN})
  set(header "${CMAKE_CURRENT_BINARY_DIR}/wayland-${P_BASENAME}-client-protocol.h")
  set(code "${CMAKE_CURRENT_BINARY_DIR}/wayland-${P_BASENAME}-protocol.c")
  add_custom_command(OUTPUT "${header}" COMMAND "${WAYLAND_SCANNER}" client-header "${P_PROTOCOL}" "${header}" DEPENDS "${P_PROTOCOL}")
  add_custom_command(OUTPUT "${code}" COMMAND "${WAYLAND_SCANNER}" private-code "${P_PROTOCOL}" "${code}" DEPENDS "${P_PROTOCOL}")
  set(${output} ${${output}} "${header}" "${code}" PARENT_SCOPE)
endfunction()
